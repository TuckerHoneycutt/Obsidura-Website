# Multi-Developer Hermes Provisioning — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a composable orchestrator CLI (`dev-admin`) that provisions per-developer Hermes containers from a declarative `developers.yaml`, with Forgejo OIDC authentication, Caddy routing, fjell setup portal, and SSH access — all with isolated per-developer data volumes.

**Architecture:** A Python Typer CLI reads `developers.yaml` and reconciles Docker state via subprocess calls to the Docker CLI and the Forgejo REST API. It creates Docker volumes, Forgejo OAuth2 apps, installs the Aurora profile, starts named Hermes containers, generates Caddy route fragments, and manages SSH authorized_keys. Fjell (Rust/Axum) serves a landing page and a setup form where developers provide their API key and SSH public key. Caddy proxies `/agent/<username>/` to each developer's Hermes dashboard, which authenticates against Forgejo OIDC.

**Tech Stack:** Python 3.13 + Typer + PyYAML, Docker CLI (subprocess), Forgejo REST API, Rust/Axum (fjell), Caddy v2.11.4, Hermes Agent v0.19.0, Docker Compose.

## Global Constraints

- All code runs inside Docker containers on `tai-review_default` network
- `HERMES_HOME=/opt/data` inside every Hermes container
- Forgejo URL: `https://superserver.tailc67a98.ts.net/git` (from `$FORGEJO_URL` env var)
- Forgejo admin token: `$FORGEJO_ADMIN_TOKEN` (required by orchestrator)
- Aurora profile URL: `$AURORA_PROFILE_URL` (Forgejo git clone URL for `aurora-agent` repo)
- Hermes image: `nousresearch/hermes-agent:latest`
- Developer containers are named `hermes-<username>`, volumes `hermes-<username>-home`
- No secrets in `developers.yaml` — all developer credentials go through fjell setup form
- Orchestrator is idempotent — running it twice produces the same state
- SSH access uses forced `docker exec` command in `authorized_keys`
- Volumes are never auto-deleted on deprovision

```python
# dev_administration/models.py
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path

import yaml


@dataclass
class DeveloperConfig:
    username: str
    display_name: str
    forgejo_user: str


@dataclass
class OrchestratorEvent:
    timestamp: str
    event_type: str
    severity: str
    developer: str | None
    message: str
    metadata: dict = field(default_factory=dict)


def parse_developers_yaml(path: Path) -> list[DeveloperConfig]:
    """Parse developers.yaml into a list of DeveloperConfig."""
    data = yaml.safe_load(path.read_text())
    raw_devs = data.get("developers", []) if data else []
    return [
        DeveloperConfig(
            username=d["username"],
            display_name=d.get("display_name", d["username"]),
            forgejo_user=d.get("forgejo_user", d["username"]),
        )
        for d in raw_devs
    ]
```

```python
# dev_administration/notifier.py
from __future__ import annotations

import json
from typing import Protocol

from dev_administration.models import OrchestratorEvent


class Notifier(Protocol):
    def notify(self, event: OrchestratorEvent) -> None: ...


class StdoutNotifier:
    def notify(self, event: OrchestratorEvent) -> None:
        print(
            f"[{event.timestamp}] {event.severity.upper()} "
            f"{event.event_type} developer={event.developer} "
            f"{event.message}"
        )


class FileNotifier:
    def __init__(self, path: str):
        self._path = path

    def notify(self, event: OrchestratorEvent) -> None:
        line = json.dumps({
            "timestamp": event.timestamp,
            "event_type": event.event_type,
            "severity": event.severity,
            "developer": event.developer,
            "message": event.message,
            "metadata": event.metadata,
        })
        with open(self._path, "a") as f:
            f.write(line + "\n")


def get_notifier(name: str, **kwargs) -> Notifier:
    if name == "stdout":
        return StdoutNotifier()
    elif name == "file":
        return FileNotifier(**kwargs)
    else:
        raise ValueError(f"Unknown notifier: {name}")
```

```python
# dev_administration/docker_utils.py
from __future__ import annotations

import subprocess

HERMES_IMAGE = "nousresearch/hermes-agent:latest"
NETWORK = "tai-review_default"


def _run(cmd: list[str], check: bool = True) -> subprocess.CompletedProcess:
    return subprocess.run(cmd, capture_output=True, text=True, check=check)


def volume_exists(name: str) -> bool:
    result = _run(["docker", "volume", "inspect", name], check=False)
    return result.returncode == 0


def create_volume(name: str) -> None:
    _run(["docker", "volume", "create", name], check=False)


def container_exists(name: str) -> bool:
    result = _run(["docker", "inspect", name], check=False)
    return result.returncode == 0


def container_status(name: str) -> str | None:
    result = _run(
        ["docker", "inspect", "-f", "{{.State.Status}}", name],
        check=False,
    )
    if result.returncode != 0:
        return None
    return result.stdout.strip()


def run_container_detached(
    name: str,
    volume: str,
    env: dict[str, str] | None = None,
    network: str = NETWORK,
    image: str = HERMES_IMAGE,
    command: str = "gateway run",
) -> str:
    cmd = [
        "docker", "run", "-d",
        "--name", name,
        "--network", network,
        "-v", f"{volume}:/opt/data",
        "-e", "HERMES_HOME=/opt/data",
        "--restart", "unless-stopped",
    ]
    if env:
        for key, value in env.items():
            cmd.extend(["-e", f"{key}={value}"])
    cmd.append(image)
    cmd.extend(command.split())
    result = _run(cmd)
    return result.stdout.strip()


def run_temp_container(
    image: str,
    command: list[str],
    volumes: dict[str, str] | None = None,
    env: dict[str, str] | None = None,
    network: str | None = None,
) -> str:
    cmd = ["docker", "run", "--rm"]
    if volumes:
        for host_path, container_path in volumes.items():
            cmd.extend(["-v", f"{host_path}:{container_path}"])
    if env:
        for key, value in env.items():
            cmd.extend(["-e", f"{key}={value}"])
    if network:
        cmd.extend(["--network", network])
    cmd.append(image)
    cmd.extend(command)
    result = _run(cmd)
    return result.stdout.strip()


def stop_and_remove_container(name: str) -> None:
    _run(["docker", "stop", name], check=False)
    _run(["docker", "rm", name], check=False)


def list_containers(prefix: str) -> list[str]:
    result = _run(
        ["docker", "ps", "-a", "--format", "{{.Names}}"],
        check=False,
    )
    if result.returncode != 0:
        return []
    return [
        name for name in result.stdout.strip().split("\n")
        if name.startswith(prefix)
    ]


def list_volumes(prefix: str) -> list[str]:
    result = _run(
        ["docker", "volume", "ls", "--format", "{{.Name}}"],
        check=False,
    )
    if result.returncode != 0:
        return []
    return [
        name for name in result.stdout.strip().split("\n")
        if name.startswith(prefix)
    ]


def docker_exec(name: str, command: str) -> str:
    result = _run(
        ["docker", "exec", name, "sh", "-c", command],
        check=False,
    )
    return result.stdout.strip()
```

```python
# dev_administration/forgejo_utils.py
from __future__ import annotations

import json
import subprocess


def _curl(url: str, token: str, method: str = "GET", data: dict | None = None) -> dict | list | None:
    cmd = [
        "curl", "-fsS",
        "-X", method,
        url,
        "-H", f"Authorization: token {token}",
        "-H", "Content-Type: application/json",
    ]
    if data:
        cmd.extend(["-d", json.dumps(data)])
    result = subprocess.run(cmd, capture_output=True, text=True, check=True)
    if not result.stdout.strip():
        return None
    return json.loads(result.stdout)


def create_oauth2_app(
    forgejo_url: str,
    token: str,
    name: str,
    redirect_uri: str,
) -> tuple[str, str]:
    """Create an OAuth2 application in Forgejo. Returns (client_id, client_secret)."""
    app = _curl(
        f"{forgejo_url}/api/v1/admin/oauth2_apps",
        token,
        method="POST",
        data={
            "name": name,
            "redirect_uris": [redirect_uri],
            "scopes": ["openid", "profile", "email"],
        },
    )
    return app["client_id"], app["client_secret"]


def find_oauth2_app(
    forgejo_url: str,
    token: str,
    name: str,
) -> int | None:
    """Find an OAuth2 app by name. Returns the app ID or None."""
    apps = _curl(
        f"{forgejo_url}/api/v1/admin/oauth2_apps",
        token,
    )
    if not apps:
        return None
    for app in apps:
        if app.get("name") == name:
            return app["id"]
    return None


def delete_oauth2_app(
    forgejo_url: str,
    token: str,
    app_id: int,
) -> None:
    """Delete an OAuth2 application by ID."""
    _curl(
        f"{forgejo_url}/api/v1/admin/oauth2_apps/{app_id}",
        token,
        method="DELETE",
    )
```

```python
from __future__ import annotations

import json
import subprocess

from dev_administration.models import DeveloperConfig


def generate_caddy_agents_conf(devs: list[DeveloperConfig], domain: str) -> str:
    """Generate Caddy route blocks for per-developer agent dashboards."""
    blocks = [
        "# Generated by dev-admin reconcile — do not edit manually",
        "",
    ]
    for dev in devs:
        name = dev.username
        blocks.extend([
            f"handle /agent/{name}/setup {{",
            "    reverse_proxy fjell:9080",
            "}",
            "",
            f"handle /agent/{name}/* {{",
            f"    uri strip_prefix /agent/{name}",
            f"    reverse_proxy hermes-{name}:9119",
            "}",
            f"handle /agent/{name} {{",
            f"    reverse_proxy hermes-{name}:9119",
            "}",
            "",
        ])
    return "\n".join(blocks)


def generate_agents_json(devs: list[DeveloperConfig]) -> str:
    """Generate the agents.json file for fjell's landing page."""
    data = [
        {"username": dev.username, "display_name": dev.display_name}
        for dev in devs
    ]
    return json.dumps(data, indent=2)


def reload_caddy(container_name: str) -> None:
    """Signal Caddy to reload its config."""
    subprocess.run(
        ["docker", "exec", container_name, "caddy", "reload",
         "--config", "/etc/caddy/Caddyfile"],
        capture_output=True, text=True, check=False,
    )
```

```python
from __future__ import annotations

from pathlib import Path


def add_ssh_key(username: str, ssh_key: str, authorized_keys_path: str) -> None:
    """Add or overwrite an SSH forced-command entry for a developer.
    Removes any existing entry for this username, then appends the new one.
    """
    remove_ssh_key(username, authorized_keys_path)
    entry = (
        f'command="docker exec -it hermes-{username} bash",'
        f'no-port-forwarding,no-X11-forwarding {ssh_key}\n'
    )
    with open(authorized_keys_path, "a") as f:
        f.write(entry)


def remove_ssh_key(username: str, authorized_keys_path: str) -> None:
    """Remove all authorized_keys entries matching hermes-<username>."""
    path = Path(authorized_keys_path)
    if not path.exists():
        return
    lines = path.read_text().splitlines()
    marker = f"hermes-{username}"
    kept = [line for line in lines if marker not in line]
    path.write_text("\n".join(kept) + ("\n" if kept else ""))
```

```python
# dev_administration/provision.py
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timezone

from dev_administration.models import DeveloperConfig, OrchestratorEvent
from dev_administration.notifier import Notifier
from dev_administration.docker_utils import (
    volume_exists, create_volume, container_exists, container_status,
    run_container_detached, run_temp_container, stop_and_remove_container,
    list_containers, docker_utils,  # for HERMES_IMAGE, NETWORK
)
from dev_administration.forgejo_utils import (
    create_oauth2_app, find_oauth2_app,
)
from dev_administration.caddy_utils import (
    generate_caddy_agents_conf, generate_agents_json, reload_caddy,
)
from dev_administration.ssh_utils import remove_ssh_key


@dataclass
class ProvisionConfig:
    forgejo_url: str
    forgego_token: str
    aurora_profile_url: str
    domain: str
    caddy_container: str
    authorized_keys_path: str


def _now() -> str:
    return datetime.now(timezone.utc).isoformat()


def _emit(
    notifier: Notifier,
    event_type: str,
    severity: str,
    developer: str | None,
    message: str,
    **metadata,
) -> OrchestratorEvent:
    event = OrchestratorEvent(
        timestamp=_now(),
        event_type=event_type,
        severity=severity,
        developer=developer,
        message=message,
        metadata=metadata,
    )
    notifier.notify(event)
    return event


def provision_developer(
    dev: DeveloperConfig,
    config: ProvisionConfig,
    notifier: Notifier,
) -> list[OrchestratorEvent]:
    """Provision a single developer's container."""
    events: list[OrchestratorEvent] = []
    name = dev.username
    vol = f"hermes-{name}-home"
    container = f"hermes-{name}"

    # 1. Volume
    if not volume_exists(vol):
        create_volume(vol)
        events.append(_emit(notifier, "volume.created", "info", name, f"Created volume {vol}"))

    # 2. OAuth2 app
    redirect_uri = f"https://{config.domain}/agent/{name}/auth/callback"
    app_id = find_oauth2_app(config.forgejo_url, config.forgego_token, f"hermes-{name}")
    if app_id is None:
        client_id, client_secret = create_oauth2_app(
            config.forgejo_url, config.forgego_token, f"hermes-{name}", redirect_uri,
        )
        events.append(_emit(notifier, "oauth2.created", "info", name, f"Created OAuth2 app hermes-{name}"))
    else:
        # App exists — we can't retrieve the secret from the API.
        # This happens on re-provision after a container was stopped.
        # The .env should already have the secret from the first provisioning.
        client_id = ""
        client_secret = ""

    # 3. Write .env (via temp container with the volume)
    env_content = f"""FORGEJO_URL={config.forgejo_url}
HERMES_HOME=/opt/data
HERMES_DASHBOARD=1
HERMES_DASHBOARD_HOST=0.0.0.0
HERMES_DASHBOARD_OIDC_ISSUER={config.forgejo_url}
HERMES_DASHBOARD_OIDC_CLIENT_ID={client_id}
HERMES_DASHBOARD_OIDC_CLIENT_SECRET={client_secret}
"""
    # Only write if client_secret was just created (can't retrieve existing)
    if client_secret:
        run_temp_container(
            image="alpine",
            command=["sh", "-c", f'cat > /opt/data/.env << "ENVEOF"\n{env_content}ENVEOF'],
            volumes={vol: "/opt/data"},
        )

    # 4. Install Aurora profile (temp Hermes container)
    run_temp_container(
        image="nousresearch/hermes-agent:latest",
        command=["hermes", "profile", "install", config.aurora_profile_url,
                  "--name", "aurora", "--force", "-y"],
        volumes={vol: "/opt/data"},
        env={"HERMES_HOME": "/opt/data"},
        network="tai-review_default",
    )

    # 5. Start persistent container
    run_container_detached(
        name=container,
        volume=vol,
        env={"HERMES_HOME": "/opt/data"},
    )

    events.append(_emit(
        notifier, "developer.provisioned", "info", name,
        f"Provisioned container {container}",
        container=container, volume=vol,
    ))
    return events


def deprovision_developer(
    username: str,
    config: ProvisionConfig,
    notifier: Notifier,
) -> list[OrchestratorEvent]:
    """Stop a developer's container. Volume is preserved."""
    events: list[OrchestratorEvent] = []
    container = f"hermes-{username}"
    vol = f"hermes-{username}-home"

    stop_and_remove_container(container)
    remove_ssh_key(username, config.authorized_keys_path)

    events.append(_emit(
        notifier, "volume.orphaned", "warning", username,
        f"Container {container} stopped. Volume {vol} preserved.",
        container=container, volume=vol,
    ))
    return events


def reconcile(
    devs: list[DeveloperConfig],
    notifier: Notifier,
    config: ProvisionConfig,
) -> list[OrchestratorEvent]:
    """Sync Docker state with developers.yaml."""
    events: list[OrchestratorEvent] = []
    desired = {dev.username for dev in devs}

    # Find actual hermes-* containers
    actual_containers = list_containers("hermes-")
    actual = {name.replace("hermes-", "", 1) for name in actual_containers}

    # Provision new developers
    for dev in devs:
        if dev.username not in actual:
            events.extend(provision_developer(dev, config, notifier))
        else:
            # Verify existing
            status = container_status(f"hermes-{dev.username}")
            if status != "running":
                events.append(_emit(
                    notifier, "container.unhealthy", "warning", dev.username,
                    f"Container hermes-{dev.username} status: {status}",
                ))

    # Deprovision removed developers
    for username in actual - desired:
        events.extend(deprovision_developer(username, config, notifier))

    # Regenerate Caddy config + agents.json
    conf = generate_caddy_agents_conf(devs, config.domain)
    agents_json = generate_agents_json(devs)

    import os
    caddy_dir = os.environ.get("CADDY_CONFIG_DIR", "/app/Caddyfile.d")
    os.makedirs(caddy_dir, exist_ok=True)
    with open(os.path.join(caddy_dir, "agents.conf"), "w") as f:
        f.write(conf)
    with open(os.path.join(caddy_dir, "agents.json"), "w") as f:
        f.write(agents_json)

    reload_caddy(config.caddy_container)

    return events
```

```python
    notifier = get_notifier("stdout")
    events = reconcile(devs, notifier, config)
    typer.echo(f"Reconciled {len(devs)} developers. {len(events)} events emitted.")


@app.command()
def status():
    """Show all developers, their containers, volumes, and health status."""
    dev_path = os.environ.get("DEVELOPERS_YAML", "developers.yaml")
    devs = parse_developers_yaml(__import__("pathlib").Path(dev_path))

    containers = {c.replace("hermes-", "", 1): c for c in list_containers("hermes-")}
    volumes = {v.replace("hermes-", "", 1).replace("-home", "", 1): v
               for v in list_volumes("hermes-")}

    typer.echo(f"\n{'Developer':<15} {'Container':<20} {'Volume':<25} {'Status':<15}")
    typer.echo(f"{'─'*15} {'─'*20} {'─'*25} {'─'*15}")

    for dev in devs:
        container = containers.get(dev.username)
        vol = volumes.get(dev.username)
        status_str = "—"
        if container:
            status_str = container_status(container) or "unknown"
        typer.echo(f"{dev.username:<15} {container or '—':<20} {vol or '—':<25} {status_str:<15}")

    # Orphaned volumes
    orphaned = set(volumes) - {d.username for d in devs}
    for name in orphaned:
        typer.echo(f"{'(orphaned)':<15} {'—':<20} {volumes[name]:<25} {'preserved':<15}")

def add(
    username: str = typer.Argument(..., help="Developer username"),
    display_name: str = typer.Option("", "--display-name", "-n", help="Display name"),
    forgejo_user: str = typer.Option("", "--forgejo-user", "-u", help="Forgejo username"),
):
    """Add a developer entry to developers.yaml. Does not provision."""
    dev_path = os.environ.get("DEVELOPERS_YAML", "developers.yaml")
    path = __import__("pathlib").Path(dev_path)

    data = yaml.safe_load(path.read_text()) if path.exists() else {"developers": []}
    if data is None:
        data = {"developers": []}
    devs = data.get("developers", [])

    if any(d["username"] == username for d in devs):
        typer.echo(f"Developer '{username}' already exists.", err=True)
        raise typer.Exit(1)

    devs.append({
        "username": username,
        "display_name": display_name or username,
        "forgejo_user": forgejo_user or username,
    })
    data["developers"] = devs
    path.write_text(yaml.dump(data, default_flow_style=False))
    typer.echo(f"Added '{username}'. Run 'dev-admin reconcile' to provision.")


@app.command()

    """Remove a developer entry from developers.yaml. Does not deprovision."""

    path = __import__("pathlib").Path(dev_path)

    data = yaml.safe_load(path.read_text())
    devs = data.get("developers", [])
    original_len = len(devs)
    devs = [d for d in devs if d["username"] != username]
    data["developers"] = devs
    path.write_text(yaml.dump(data, default_flow_style=False))

    if len(devs) == original_len:
        typer.echo(f"Developer '{username}' not found.", err=True)
        raise typer.Exit(1)
    typer.echo(f"Removed '{username}'. Run 'dev-admin reconcile' to deprovision.")
```

```rust
// src/routes/setup.rs
use axum::{
    extract::Path,
    response::{Html, Redirect},
    routing::{get, post},
    Form,
};
use serde::Deserialize;

#[derive(Deserialize)]
pub struct SetupForm {
    pub api_key: String,
    pub ssh_key: Option<String>,
}

pub async fn setup_form(Path(username): Path<String>) -> Html<String> {
    let html = format!(r#"<!DOCTYPE html>
<html><head><title>Setup — {username}</title></head>
<body style="font-family:system-ui;max-width:500px;margin:80px auto">
<h1>Setup: {username}</h1>
<form method="POST" action="/agent/{username}/setup">
  <p><label>OpenRouter API Key:<br>
    <input type="password" name="api_key" required style="width:100%;padding:8px">
  </label></p>
  <p><label>SSH Public Key (optional):<br>
    <textarea name="ssh_key" rows="4" style="width:100%;padding:8px"
      placeholder="ssh-ed25519 AAAA... your@laptop"></textarea>
  </label></p>
  <p><button type="submit" style="padding:10px 20px">Save</button></p>
</form>
<p><small>These values overwrite any previous submission. You can return to this page anytime to update.</small></p>
</body></html>"#, username = username);
    Html(html)
}

pub async fn setup_submit(
    Path(username): Path<String>,
    Form(form): Form<SetupForm>,
) -> Redirect {
    let container = format!("hermes-{}", username);

    // Write API key to .env (delete old, append new)
    let env_cmd = format!(
        "sed -i '/^OPENROUTER_API_KEY=/d' /opt/data/.env; echo 'OPENROUTER_API_KEY={}' >> /opt/data/.env",
        form.api_key
    );
    let _ = std::process::Command::new("docker")
        .args(["exec", &container, "sh", "-c", &env_cmd])
        .output();

    // Write SSH key to authorized_keys if provided
    if let Some(ssh_key) = form.ssh_key {
        if !ssh_key.trim().is_empty() {
            let auth_path = std::env::var("AUTHORIZED_KEYS_PATH")
                .unwrap_or_else(|_| "/app/authorized_keys".to_string());

            // Remove old entry for this user
            if let Ok(content) = std::fs::read_to_string(&auth_path) {
                let marker = format!("hermes-{}", username);
                let kept: String = content
                    .lines()
                    .filter(|line| !line.contains(&marker))
                    .map(|line| format!("{}\n", line))
                    .collect();
                let _ = std::fs::write(&auth_path, kept);
            }

            // Append new entry
            let entry = format!(
                r#"command="docker exec -it hermes-{} bash",no-port-forwarding,no-X11-forwarding {}"#,
                username, ssh_key.trim()
            );
            let _ = std::fs::OpenOptions::new()
                .append(true)
                .open(&auth_path)
                .and_then(|mut f| std::io::Write::write_all(&mut f, format!("{}\n", entry).as_bytes()));
        }
    }

    Redirect::to(&format!("/agent/{}/", username))
}

pub fn router() -> axum::Router {
    axum::Router::new()
        .route("/agent/:username/setup", get(setup_form).post(setup_submit))
}
```
